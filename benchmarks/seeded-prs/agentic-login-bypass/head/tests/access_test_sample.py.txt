from billing.access import Invoice, User, can_view_invoice


def test_owner_can_view_invoice() -> None:
    """Owners can read their own invoice."""

    user = User(id="acct_123", role="member")
    invoice = Invoice(id="inv_1", owner_id="acct_123", amount_cents=4200)

    assert can_view_invoice(user, invoice, {}) is True


def test_other_member_cannot_view_invoice() -> None:
    """Members cannot read another account's invoice."""

    user = User(id="acct_999", role="member")
    invoice = Invoice(id="inv_1", owner_id="acct_123", amount_cents=4200)

    assert can_view_invoice(user, invoice, {}) is False
